Creating 3D Plot

library(plotly)
advertising |>
  plot_ly(x=~TV,y=~radio,z=~sales) |>
  add_markers(size=3,color=I("black"))
x_lims=c(0,300) #Limits for x-variable
y_lims=c(0,50) #Limits for y-variable
xy_grid=expand.grid(TV=x_lims,radio=y_lims) #pairs of x-y values
yhat=predict(sales_lm_reduced,xy_grid) #Calculates predictions (z)
z=matrix(yhat,nrow=2,ncol=2) #Turns predictions into matrix
advertising |>
  plot_ly(x=~TV,y=~radio,z=~sales) |>
  add_markers(size=3,color=I("black")) |>
  add_surface(x=~x_lims,y=~y_lims,z=~t(z),colorscale="Portland")
